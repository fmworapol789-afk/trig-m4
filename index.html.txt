  <script>
    const canvas = document.getElementById('pythaCanvas');
    const ctx = canvas.getContext('2d');

    let sideA = 3;
    let sideB = 4;
    let sideC = 5;
    let tiles = [];
    let progress = 0; // 0 ถึง 1
    let animId = null;

    // ตั้งค่าชุดสามเหลี่ยม
    function setTriangle(a, b, btn) {
      if (btn) {
        document.querySelectorAll('.preset-btn').forEach(el => el.classList.remove('active'));
        btn.classList.add('active');
      }
      sideA = a;
      sideB = b;
      sideC = Math.round(Math.sqrt(a * a + b * b));

      document.getElementById('lenA').innerText = sideA;
      document.getElementById('lenB').innerText = sideB;
      document.getElementById('lenC').innerText = sideC;
      document.getElementById('areaA').innerText = (sideA * sideA) + ' ช่อง';
      document.getElementById('areaB').innerText = (sideB * sideB) + ' ช่อง';
      document.getElementById('areaC').innerText = (sideC * sideC);
      document.getElementById('summaryFormula').innerText =
        `${sideA*sideA} + ${sideB*sideB} = ${sideC*sideC} ช่อง (${sideA}² + ${sideB}² = ${sideC}²)`;

      resetPuzzle();
    }

    // สร้างตำแหน่งเริ่มต้นและปลายทางของช่องตารางแต่ละชิ้น
    function buildTiles() {
      tiles = [];
      const scale = sideA === 3 ? 28 : 14; // ปรับขนาดช่องตามชุดตัวเลข
      const originX = 185;
      const originY = 250;

      // พิกัดมุมทั้งสามของรูปสามเหลี่ยมมุมฉาก (มุมฉากอยู่ที่ originX, originY)
      const pRight = { x: originX, y: originY };
      const pTop = { x: originX, y: originY - sideA * scale };
      const pBottomRight = { x: originX + sideB * scale, y: originY };

      // 1. สร้างช่องตารางบนด้านสั้น (a) สีฟ้า (วางด้านซ้ายของสามเหลี่ยม)
      let targetSlots = [];
      // คำนวณช่องปลายทางบนด้านตรงข้ามมุมฉาก (c)
      const dx = pBottomRight.x - pTop.x;
      const dy = pBottomRight.y - pTop.y;
      const len = Math.hypot(dx, dy);
      const ux = dx / len; // เวกเตอร์ตามแนวด้าน c
      const uy = dy / len;
      const nx = uy;       // เวกเตอร์ตั้งฉากพุ่งออกด้านนอก
      const ny = -ux;

      for (let r = 0; r < sideC; r++) {
        for (let c = 0; c < sideC; c++) {
          targetSlots.push({
            x: pTop.x + (c * scale) * ux + (r * scale) * nx,
            y: pTop.y + (c * scale) * uy + (r * scale) * ny,
            angle: Math.atan2(uy, ux)
          });
        }
      }

      let slotIndex = 0;

      // ช่องจากด้าน a (สีฟ้า)
      for (let r = 0; r < sideA; r++) {
        for (let c = 0; c < sideA; c++) {
          tiles.push({
            startX: pTop.x - (sideA - c) * scale,
            startY: pTop.y + r * scale,
            startAngle: 0,
            endX: targetSlots[slotIndex].x,
            endY: targetSlots[slotIndex].y,
            endAngle: targetSlots[slotIndex].angle,
            size: scale,
            color: '#3b82f6',
            border: '#1d4ed8',
            index: slotIndex
          });
          slotIndex++;
        }
      }

      // ช่องจากด้าน b (สีเหลืองส้ม)
      for (let r = 0; r < sideB; r++) {
        for (let c = 0; c < sideB; c++) {
          tiles.push({
            startX: originX + c * scale,
            startY: originY + r * scale,
            startAngle: 0,
            endX: targetSlots[slotIndex].x,
            endY: targetSlots[slotIndex].y,
            endAngle: targetSlots[slotIndex].angle,
            size: scale,
            color: '#f59e0b',
            border: '#b45309',
            index: slotIndex
          });
          slotIndex++;
        }
      }

      return { pRight, pTop, pBottomRight, scale, ux, uy, nx, ny };
    }

    function draw() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      const geom = buildTiles();
      const { pRight, pTop, pBottomRight, scale, ux, uy, nx, ny } = geom;

      // วาดกรอบช่องว่างต้นทาง (a และ b)
      ctx.strokeStyle = '#93c5fd';
      ctx.lineWidth = 1.5;
      ctx.strokeRect(pTop.x - sideA * scale, pTop.y, sideA * scale, sideA * scale);

      ctx.strokeStyle = '#fcd34d';
      ctx.strokeRect(pRight.x, pRight.y, sideB * scale, sideB * scale);

      // วาดกรอบเป้าหมาย "ด้านพี่ใหญ่ (c)" สีแดง
      ctx.save();
      ctx.translate(pTop.x, pTop.y);
      ctx.rotate(Math.atan2(uy, ux));
      ctx.fillStyle = 'rgba(239, 68, 68, 0.08)';
      ctx.fillRect(0, -sideC * scale, sideC * scale, sideC * scale);
      ctx.strokeStyle = '#ef4444';
      ctx.lineWidth = 2.5;
      ctx.strokeRect(0, -sideC * scale, sideC * scale, sideC * scale);

      // วาดเส้นตารางจาง ๆ ในกรอบพี่ใหญ่
      ctx.strokeStyle = 'rgba(239, 68, 68, 0.25)';
      ctx.lineWidth = 1;
      for (let i = 1; i < sideC; i++) {
        ctx.beginPath();
        ctx.moveTo(i * scale, 0);
        ctx.lineTo(i * scale, -sideC * scale);
        ctx.stroke();

        ctx.beginPath();
        ctx.moveTo(0, -i * scale);
        ctx.lineTo(sideC * scale, -i * scale);
        ctx.stroke();
      }
      ctx.restore();

      // วาดรูปสามเหลี่ยมมุมฉากตรงกลาง
      ctx.beginPath();
      ctx.moveTo(pRight.x, pRight.y);
      ctx.lineTo(pTop.x, pTop.y);
      ctx.lineTo(pBottomRight.x, pBottomRight.y);
      ctx.closePath();
      ctx.fillStyle = '#e2e8f0';
      ctx.fill();
      ctx.strokeStyle = '#334155';
      ctx.lineWidth = 2.5;
      ctx.stroke();

      // วาดสัญลักษณ์มุมฉากสีแดงชัดเจน
      const boxSize = 16;
      ctx.strokeStyle = '#ef4444';
      ctx.lineWidth = 2.5;
      ctx.strokeRect(pRight.x, pRight.y - boxSize, boxSize, boxSize);

      // วาดชิ้นส่วนจิ๊กซอว์ทีละชิ้นตามค่า progress
      const totalTiles = tiles.length;
      let filled = 0;

      tiles.forEach((t, idx) => {
        // แบ่งจังหวะการเคลื่อนที่ให้ทยอยเรียงเข้าไปทีละชิ้นอย่างสวยงาม
        const stepStart = (idx / totalTiles) * 0.65;
        const stepEnd = stepStart + 0.35;
        let localP = (progress - stepStart) / (stepEnd - stepStart);
        localP = Math.max(0, Math.min(1, localP));

        // ฟังก์ชัน Ease-in-out
        const ease = localP < 0.5 ? 2 * localP * localP : -1 + (4 - 2 * localP) * localP;

        if (localP >= 0.95) filled++;

        const curX = t.startX + (t.endX - t.startX) * ease;
        const curY = t.startY + (t.endY - t.startY) * ease;
        const curAngle = t.startAngle + (t.endAngle - t.startAngle) * ease;

        ctx.save();
        ctx.translate(curX, curY);
        ctx.rotate(curAngle);
        ctx.fillStyle = t.color;
        ctx.strokeStyle = '#ffffff';
        ctx.lineWidth = 1.2;

        // ชดเชยแกน Y เมื่อหมุนเข้าสู่กรอบด้าน c
        const offsetY = localP > 0 ? -t.size * ease : 0;
        ctx.fillRect(0, offsetY, t.size, t.size);
        ctx.strokeRect(0, offsetY, t.size, t.size);
        ctx.restore();
      });

      // ป้ายข้อความกำกับด้านพี่ใหญ่
      ctx.save();
      ctx.fillStyle = '#b91c1c';
      ctx.font = 'bold 14px Prompt, sans-serif';
      ctx.fillText('ด้านพี่ใหญ่ (ตรงข้ามมุมฉาก)', pTop.x + (sideC * scale * 0.35), pTop.y - 12);
      ctx.restore();

      // อัปเดตตัวเลขหน้าจอ
      document.getElementById('filledCount').innerText = filled;
      const banner = document.getElementById('successBanner');
      if (filled === totalTiles) {
        banner.style.display = 'block';
        document.getElementById('hintText').innerText = 'ครบแล้ว! ลองสังเกตผลรวมช่องด้านล่าง';
      } else {
        banner.style.display = 'none';
        document.getElementById('hintText').innerText = 'แตะที่หน้าจอ หรือเลื่อนแถบด้านล่างเพื่อย้ายช่อง';
      }
    }

    function onSliderChange(val) {
      cancelAnimationFrame(animId);
      progress = val / 100;
      document.getElementById('percentText').innerText = Math.round(val) + '%';
      draw();
    }

    function resetPuzzle() {
      cancelAnimationFrame(animId);
      progress = 0;
      document.getElementById('progressSlider').value = 0;
      document.getElementById('percentText').innerText = '0%';
      draw();
    }

    function animateFill() {
      cancelAnimationFrame(animId);
      if (progress >= 0.99) progress = 0;

      function step() {
        progress += 0.015;
        if (progress >= 1) {
          progress = 1;
          document.getElementById('progressSlider').value = 100;
          document.getElementById('percentText').innerText = '100%';
          draw();
          return;
        }
        document.getElementById('progressSlider').value = Math.round(progress * 100);
        document.getElementById('percentText').innerText = Math.round(progress * 100) + '%';
        draw();
        animId = requestAnimationFrame(step);
      }
      animId = requestAnimationFrame(step);
    }

    // แตะที่รูปภาพเพื่อสลับย้าย/รีเซ็ตได้ทันทีบนมือถือ
    canvas.addEventListener('click', () => {
      if (progress < 1) {
        animateFill();
      } else {
        resetPuzzle();
      }
    });

    // เริ่มต้นวาดภาพชุดแรก
    setTriangle(3, 4, null);
  </script>
